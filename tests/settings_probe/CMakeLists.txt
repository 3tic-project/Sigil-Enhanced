cmake_minimum_required(VERSION 3.18)
project(SigilSettingsProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
    add_compile_options(/utf-8)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
enable_testing()

add_executable(settings_concurrency_test ../settings_concurrency_test.cpp)
target_link_libraries(settings_concurrency_test PRIVATE Qt6::Core)
add_test(NAME settings_concurrency COMMAND settings_concurrency_test)
set_tests_properties(settings_concurrency PROPERTIES TIMEOUT 60)

add_executable(window_startup_policy_test ../window_startup_policy_test.cpp)
target_include_directories(window_startup_policy_test PRIVATE ../../src)
target_link_libraries(window_startup_policy_test PRIVATE Qt6::Widgets)
add_test(NAME window_startup_policy COMMAND window_startup_policy_test)
set_tests_properties(window_startup_policy PROPERTIES TIMEOUT 15)

add_executable(file_dialog_policy_test ../file_dialog_policy_test.cpp)
target_include_directories(file_dialog_policy_test PRIVATE ../../src)
target_link_libraries(file_dialog_policy_test PRIVATE Qt6::Widgets)
add_test(NAME file_dialog_policy COMMAND file_dialog_policy_test)
set_tests_properties(file_dialog_policy PROPERTIES TIMEOUT 15)
